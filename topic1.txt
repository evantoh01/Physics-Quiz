    <script>
        const answeredQuestions = new Set();
        const totalQuestions = 58;
        let correctCount = 0;

        function checkAnswer(qNum) {
            const selectedCard = document.querySelector(`input[name="q${qNum}"]:checked`);
            const feedbackDiv = document.getElementById(`q${qNum}-feedback`);
            const cardElement = document.getElementById(`q${qNum}-card`);
            const correctAnswer = cardElement.getAttribute('data-answer');

            if (!selectedCard) {
                feedbackDiv.className = "feedback incorrect";
                feedbackDiv.innerHTML = "Please select an option before submitting!";
                return;
            }

            const userChoice = selectedCard.value;

            if (!answeredQuestions.has(qNum)) {
                answeredQuestions.add(qNum);
            }

            recalculateScore();

            if (userChoice === correctAnswer) {
                feedbackDiv.className = "feedback correct";
                feedbackDiv.innerHTML = `<strong>Correct!</strong> Option ${correctAnswer} is the right answer.`;
            } else {
                feedbackDiv.className = "feedback incorrect";
                feedbackDiv.innerHTML = `<strong>Incorrect.</strong> The correct answer is Option ${correctAnswer}.`;
            }
        }

        function recalculateScore() {
            let currentCorrect = 0;
            let currentAttempted = answeredQuestions.size;

            answeredQuestions.forEach(qNum => {
                const cardEl = document.getElementById(`q${qNum}-card`);
                const correctAns = cardEl.getAttribute('data-answer');
                const radios = document.querySelectorAll(`input[name="q${qNum}"]`);
                let selectedVal = null;
                radios.forEach(r => { if (r.checked) selectedVal = r.value; });
                if (selectedVal === correctAns) {
                    currentCorrect++;
                }
            });

            correctCount = currentCorrect;
            const percentage = totalQuestions > 0 ? Math.round((correctCount / totalQuestions) * 100) : 0;

            document.getElementById('attempted-count').innerText = `${currentAttempted} / ${totalQuestions}`;
            document.getElementById('current-score').innerText = correctCount;
            document.getElementById('percentage-score').innerText = `${percentage}%`;
        }
    </script>
